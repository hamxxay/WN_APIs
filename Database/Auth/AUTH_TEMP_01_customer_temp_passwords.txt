-- ============================================================================================
-- AUTH_TEMP_01: temporary passwords for customers (CMS-created customers can log in to worknestpk.com / the app).
-- Run as ONE statement in DBeaver (select all, then Ctrl+Enter), on the test database and on the live database.
-- One transaction: if anything fails, nothing is changed. Safe to run again. Non-destructive: no row is deleted or reset.
--
-- Every customer gets a temporary password (staff see it once and the customer is e-mailed); at the first sign-in the customer
-- must set their own. Signing in with Google (same email) counts as activated. Until this script has run the staff endpoints
-- answer 503 "Database update needed" and sign-in works as before.
--
--   dbo.WN_Users.MustChangePassword     bit NOT NULL default 0: 1 while a temporary password is pending.
--   dbo.WN_Users.TempPasswordSetAt      datetime2 NULL (UTC): when it was handed out.
--   dbo.WN_Users.TempPassword           nvarchar(64) NULL: the readable temporary password. DELIBERATE OWNER DECISION: staff can look
--                                       it up while the login is pending; it is cleared (NULL) the moment the customer sets their
--                                       own password or signs in with Google. PasswordHash holds its PBKDF2 hash; login uses that.
--   dbo.WN_Users_SetTempPassword        sets hash + readable password + flag for a general-role user.
--   dbo.WN_Customers_EnsureLoginUser    finds / creates (general role) the customer's login user and links WN_Customers.UserId.
--   dbo.WN_Users_ChangePassword         the user's own new hash; clears flag and readable password.
--   dbo.WN_Users_ActivateLogin          Google / Firebase sign-in: clears flag and readable password and drops the temporary hash.
--   dbo.WN_Customers_LoginStatus        per customer: user, role, flag, TempPassword, HasSignedIn (LOGIN row in WN_LoginHistory).
--   dbo.WN_Users_GetByEmail             now returns MustChangePassword; lists its columns instead of u.* so TempPassword never leaves.
--   dbo.TR_WN_Users_Audit               the audit trigger copied whole rows into WN_AuditLog: it now lists the columns and leaves out
--                                       PasswordHash and TempPassword (old audit rows are not touched).
-- ============================================================================================
SET XACT_ABORT ON;
SET QUOTED_IDENTIFIER ON;
SET ANSI_NULLS ON;
BEGIN TRY
    IF OBJECT_ID(N'dbo.WN_Users', N'U') IS NULL OR OBJECT_ID(N'dbo.WN_Customers', N'U') IS NULL
        THROW 50001, N'dbo.WN_Users / dbo.WN_Customers is missing: wrong database? Nothing changed.', 1;

    BEGIN TRANSACTION;

    -- 1. Columns.
    IF COL_LENGTH(N'dbo.WN_Users', N'MustChangePassword') IS NULL
        EXEC(N'ALTER TABLE dbo.WN_Users ADD MustChangePassword BIT NOT NULL CONSTRAINT DF_WN_Users_MustChangePassword DEFAULT (0);');
    IF COL_LENGTH(N'dbo.WN_Users', N'TempPasswordSetAt') IS NULL
        EXEC(N'ALTER TABLE dbo.WN_Users ADD TempPasswordSetAt DATETIME2(3) NULL;');
    IF COL_LENGTH(N'dbo.WN_Users', N'TempPassword') IS NULL
        EXEC(N'ALTER TABLE dbo.WN_Users ADD TempPassword NVARCHAR(64) NULL;');

    -- 2. Procedures.
    EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_Users_SetTempPassword
    @UserId       INT,
    @PasswordHash NVARCHAR(1000),   -- pbkdf2$... : what login checks
    @TempPassword NVARCHAR(64)      -- the readable temporary password (owner decision), cleared when the customer sets their own
AS
BEGIN
    -- AUTH_TEMP_01: gives a customer (general role) a temporary password and makes them change it at the next sign-in.
    -- Only general-role users are touched: AffectedRows = 0 for staff accounts.
    SET NOCOUNT ON;
    UPDATE dbo.WN_Users
       SET PasswordHash = @PasswordHash, TempPassword = @TempPassword, MustChangePassword = 1,
           TempPasswordSetAt = SYSUTCDATETIME(), UpdatedOn = GETDATE()
     WHERE Id = @UserId AND RoleId = 14;
    SELECT @@ROWCOUNT AS AffectedRows;
END;');

    EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_Customers_EnsureLoginUser
    @CustomerId INT
AS
BEGIN
    -- AUTH_TEMP_01: the customer''s login user (UserId, RoleId, IsActive), created with the general role (14) when there is none and
    -- linked to WN_Customers.UserId. UserId is NULL when the customer has no email. An existing user is never changed here.
    SET NOCOUNT ON;
    DECLARE @Email NVARCHAR(255), @First NVARCHAR(100), @Last NVARCHAR(100), @Phone NVARCHAR(50), @LinkedUserId INT, @UserId INT;

    SELECT @Email = LTRIM(RTRIM(c.Email)), @First = c.FirstName, @Last = c.LastName, @Phone = c.PhoneNumber, @LinkedUserId = c.UserId
      FROM dbo.WN_Customers c WHERE c.Id = @CustomerId;

    IF @Email IS NULL OR @Email = N''''
    BEGIN
        SELECT CAST(NULL AS INT) AS UserId, CAST(NULL AS INT) AS RoleId, CAST(0 AS BIT) AS IsActive;
        RETURN;
    END;

    SELECT TOP 1 @UserId = u.Id FROM dbo.WN_Users u
     WHERE u.Id = @LinkedUserId OR LOWER(LTRIM(RTRIM(u.Email))) = LOWER(@Email)
     ORDER BY CASE WHEN u.Id = @LinkedUserId THEN 0 ELSE 1 END, u.IsActive DESC, u.Id DESC;

    IF @UserId IS NULL
    BEGIN
        INSERT INTO dbo.WN_Users (IdGUID, Email, UserName, Name, PhoneNumber, RoleId, IsActive, CreatedOn)
        VALUES (NEWID(), @Email, @Email, LTRIM(RTRIM(ISNULL(@First, N'''') + N'' '' + ISNULL(@Last, N''''))), @Phone, 14, 1, GETDATE());
        SET @UserId = CAST(SCOPE_IDENTITY() AS INT);
    END;

    UPDATE dbo.WN_Customers SET UserId = @UserId
     WHERE Id = @CustomerId AND (UserId IS NULL OR UserId <= 0)
       AND NOT EXISTS (SELECT 1 FROM dbo.WN_Customers c2 WITH (NOLOCK) WHERE c2.UserId = @UserId);

    SELECT u.Id AS UserId, u.RoleId, CAST(ISNULL(u.IsActive, 0) AS BIT) AS IsActive FROM dbo.WN_Users u WHERE u.Id = @UserId;
END;');

    EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_Users_ChangePassword
    @UserId       INT,
    @PasswordHash NVARCHAR(1000)
AS
BEGIN
    -- AUTH_TEMP_01: the user chose their own password: store its hash, clear the flag and the readable temporary password.
    SET NOCOUNT ON;
    UPDATE dbo.WN_Users
       SET PasswordHash = @PasswordHash, MustChangePassword = 0, TempPassword = NULL, UpdatedOn = GETDATE()
     WHERE Id = @UserId;
    SELECT @@ROWCOUNT AS AffectedRows;
END;');

    EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_Users_ActivateLogin
    @UserId INT
AS
BEGIN
    -- AUTH_TEMP_01: Google / Firebase proved the customer owns the email. Only while a temporary password is pending: clear the
    -- flag and the readable temporary password, and drop the temporary password hash so it stops working.
    SET NOCOUNT ON;
    UPDATE dbo.WN_Users
       SET MustChangePassword = 0, TempPassword = NULL, PasswordHash = NULL, UpdatedOn = GETDATE()
     WHERE Id = @UserId AND MustChangePassword = 1;
    SELECT @@ROWCOUNT AS AffectedRows;
END;');

    EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_Customers_LoginStatus
    @CustomerIds NVARCHAR(MAX) = NULL   -- comma separated customer ids; NULL = every customer
AS
BEGIN
    -- AUTH_TEMP_01: login state of customers. The user is the one linked by WN_Customers.UserId, else the one with the same email.
    -- HasSignedIn = at least one LOGIN row in WN_LoginHistory. TempPassword is the readable temporary password (owner decision);
    -- the API shows it only to admin / super admin and only while the login is pending.
    SET NOCOUNT ON;
    SELECT c.Id AS CustomerId, c.FirstName, c.LastName, c.Company, c.Email,
           CAST(ISNULL(c.IsActive, 1) AS BIT) AS CustomerActive,
           u.Id AS UserId, u.RoleId,
           CAST(ISNULL(u.IsActive, 0) AS BIT) AS UserActive,
           CAST(ISNULL(u.MustChangePassword, 0) AS BIT) AS MustChangePassword,
           u.TempPassword,
           CAST(CASE WHEN u.Id IS NOT NULL AND EXISTS (SELECT 1 FROM dbo.WN_LoginHistory h WITH (NOLOCK)
                                                        WHERE h.UserId = u.Id AND h.EventType = ''LOGIN'')
                     THEN 1 ELSE 0 END AS BIT) AS HasSignedIn
      FROM dbo.WN_Customers c
      OUTER APPLY (SELECT TOP 1 x.Id, x.RoleId, x.IsActive, x.MustChangePassword, x.TempPassword
                     FROM dbo.WN_Users x
                    WHERE x.Id = c.UserId
                       OR (c.Email IS NOT NULL AND LTRIM(RTRIM(c.Email)) <> N'''' AND LOWER(LTRIM(RTRIM(x.Email))) = LOWER(LTRIM(RTRIM(c.Email))))
                    ORDER BY CASE WHEN x.Id = c.UserId THEN 0 ELSE 1 END, x.IsActive DESC, x.Id DESC) u
     WHERE @CustomerIds IS NULL
        OR c.Id IN (SELECT TRY_CAST(LTRIM(RTRIM(value)) AS INT) FROM STRING_SPLIT(@CustomerIds, N'',''))
     ORDER BY c.Id;
END;');

    EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_Users_GetByEmail
    @Email NVARCHAR(256)
AS
BEGIN
    -- AUTH_TEMP_01: as before (u.* plus the customer), now with MustChangePassword and without TempPassword / TempPasswordSetAt
    -- (the API only needs the flag; staff read the readable temporary password through WN_Customers_LoginStatus).
    SET NOCOUNT ON;

    SELECT
        u.Id, u.IdGUID, u.Name, u.UserName, u.RoleId, u.Email, u.PhoneNumber, u.LockoutEnd, u.LockoutEnabled,
        u.CreatedOn, u.UpdatedOn, u.CompanyId, u.Status, u.PasswordHash, u.IsActive, u.Address, u.CnicOrPassport,
        u.AvatarUrl, u.Notes, u.CreatedById, u.CityId, u.LocationId, u.MustChangePassword,
        c.Id        AS CustomerId,
        c.IdGUID    AS CustomerGuid
    FROM dbo.WN_Users u
    LEFT JOIN dbo.WN_Customers c ON c.Email = u.Email AND c.IsActive = 1
    WHERE u.Email = @Email
      AND u.IsActive = 1;
END;');

    -- 3. Audit trigger without the password columns.
    EXEC(N'CREATE OR ALTER TRIGGER dbo.TR_WN_Users_Audit
ON dbo.WN_Users
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    -- AUTH_TEMP_01: as before, but the audit JSON lists the columns and leaves out PasswordHash and TempPassword.
    SET NOCOUNT ON;

    IF NOT EXISTS (SELECT 1 FROM inserted) AND NOT EXISTS (SELECT 1 FROM deleted)
        RETURN;

    DECLARE @UserId INT = CAST(SESSION_CONTEXT(N''AppUserId'') AS INT);
    IF @UserId IS NULL SET @UserId = -1;

    DECLARE @UserEmail NVARCHAR(256) = CAST(SESSION_CONTEXT(N''AppUserEmail'') AS NVARCHAR(256));
    DECLARE @SourceApp NVARCHAR(50) = CAST(SESSION_CONTEXT(N''SourceApp'') AS NVARCHAR(50));
    DECLARE @EntityName NVARCHAR(128) = N''WN_Users'';
    DECLARE @SysUtc DATETIME2(3) = SYSUTCDATETIME();

    BEGIN TRY
        IF EXISTS (SELECT 1 FROM inserted) AND NOT EXISTS (SELECT 1 FROM deleted)
        BEGIN
            INSERT INTO dbo.WN_AuditLog (
                EntityName, RecordId, Operation, OldValues, NewValues,
                ErrorMessage, ErrorSeverity, StackTrace, ChangedByUserId, ChangedByUserEmail, ChangedAt, SourceApp
            )
            SELECT @EntityName, CAST(i.[Id] AS NVARCHAR(64)), ''I'', NULL,
                (SELECT i.Id, i.IdGUID, i.Name, i.UserName, i.RoleId, i.Email, i.PhoneNumber, i.LockoutEnd, i.LockoutEnabled, i.CreatedOn, i.UpdatedOn, i.CompanyId, i.Status, i.IsActive, i.Address, i.CnicOrPassport, i.AvatarUrl, i.Notes, i.CreatedById, i.CityId, i.LocationId, i.MustChangePassword, i.TempPasswordSetAt FOR JSON PATH, WITHOUT_ARRAY_WRAPPER),
                NULL, NULL, NULL, @UserId, @UserEmail, @SysUtc, @SourceApp
            FROM inserted i;
        END

        ELSE IF EXISTS (SELECT 1 FROM inserted) AND EXISTS (SELECT 1 FROM deleted)
        BEGIN
            INSERT INTO dbo.WN_AuditLog (
                EntityName, RecordId, Operation, OldValues, NewValues,
                ErrorMessage, ErrorSeverity, StackTrace, ChangedByUserId, ChangedByUserEmail, ChangedAt, SourceApp
            )
            SELECT @EntityName, CAST(i.[Id] AS NVARCHAR(64)), ''U'',
                (SELECT d.Id, d.IdGUID, d.Name, d.UserName, d.RoleId, d.Email, d.PhoneNumber, d.LockoutEnd, d.LockoutEnabled, d.CreatedOn, d.UpdatedOn, d.CompanyId, d.Status, d.IsActive, d.Address, d.CnicOrPassport, d.AvatarUrl, d.Notes, d.CreatedById, d.CityId, d.LocationId, d.MustChangePassword, d.TempPasswordSetAt FOR JSON PATH, WITHOUT_ARRAY_WRAPPER),
                (SELECT i.Id, i.IdGUID, i.Name, i.UserName, i.RoleId, i.Email, i.PhoneNumber, i.LockoutEnd, i.LockoutEnabled, i.CreatedOn, i.UpdatedOn, i.CompanyId, i.Status, i.IsActive, i.Address, i.CnicOrPassport, i.AvatarUrl, i.Notes, i.CreatedById, i.CityId, i.LocationId, i.MustChangePassword, i.TempPasswordSetAt FOR JSON PATH, WITHOUT_ARRAY_WRAPPER),
                NULL, NULL, NULL, @UserId, @UserEmail, @SysUtc, @SourceApp
            FROM inserted i
            INNER JOIN deleted d ON i.[Id] = d.[Id];
        END

        ELSE IF NOT EXISTS (SELECT 1 FROM inserted) AND EXISTS (SELECT 1 FROM deleted)
        BEGIN
            INSERT INTO dbo.WN_AuditLog (
                EntityName, RecordId, Operation, OldValues, NewValues,
                ErrorMessage, ErrorSeverity, StackTrace, ChangedByUserId, ChangedByUserEmail, ChangedAt, SourceApp
            )
            SELECT @EntityName, CAST(d.[Id] AS NVARCHAR(64)), ''D'',
                (SELECT d.Id, d.IdGUID, d.Name, d.UserName, d.RoleId, d.Email, d.PhoneNumber, d.LockoutEnd, d.LockoutEnabled, d.CreatedOn, d.UpdatedOn, d.CompanyId, d.Status, d.IsActive, d.Address, d.CnicOrPassport, d.AvatarUrl, d.Notes, d.CreatedById, d.CityId, d.LocationId, d.MustChangePassword, d.TempPasswordSetAt FOR JSON PATH, WITHOUT_ARRAY_WRAPPER),
                NULL, NULL, NULL, NULL, @UserId, @UserEmail, @SysUtc, @SourceApp
            FROM deleted d;
        END
    END TRY
    BEGIN CATCH
        BEGIN TRY
            INSERT INTO dbo.WN_AuditLog (
                EntityName, RecordId, Operation, OldValues, NewValues,
                ErrorMessage, ErrorSeverity, StackTrace,
                ChangedByUserId, ChangedByUserEmail, ChangedAt, SourceApp
            )
            VALUES (
                @EntityName, ''TRIGGER_ERROR'', ''E'', NULL, NULL,
                ERROR_MESSAGE(), ''Error'',
                CONCAT(''Line: '', ERROR_LINE(), '' State: '', ERROR_STATE(), '' Procedure: '', ERROR_PROCEDURE()),
                @UserId, @UserEmail, SYSUTCDATETIME(), @SourceApp
            );
        END TRY
        BEGIN CATCH
            -- Silent swallow to guarantee original DML never fails
        END CATCH
    END CATCH
END;');

    COMMIT TRANSACTION;
    SELECT N'OK: customer temporary passwords installed (AUTH_TEMP_01).' AS Result;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;
END CATCH
