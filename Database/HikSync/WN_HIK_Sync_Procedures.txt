-- ============================================================================================
-- WN_HIK_Sync_* procedures for the Hikvision sync built into WN_APIs (HikSyncService, HikSync:Enabled).
-- They run the same queries the HIK_Access-Control Node scheduler runs (same WN_HIK_* tables and columns).
-- 31 procedures, CREATE OR ALTER: no table is created or changed. Safe to run again.
-- Run on the database the API uses (test, then live) as ONE statement in DBeaver.
-- Needs SQL Server 2016+ with compatibility level 130+ (WN_HIK_Sync_Users_Rebuild reads JSON).
-- ============================================================================================
SET NOCOUNT ON;

BEGIN
    SELECT 'Database compatibility level is below 130 (needed for OPENJSON). Nothing was changed - tell Claude.' AS Result;
    RETURN;
END

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_Devices_Get
    @DeviceId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;
    -- Machines (with group name) for the sync jobs: entrances first, then meeting rooms, then rooms by number.
    SELECT d.Id, d.Device_Name, d.Host, d.Host2, d.Port, d.Use_https, d.Username, d.Password, d.Online, d.Code, g.Name AS GroupName
    FROM dbo.WN_HIK_Devices d WITH (NOLOCK)
    LEFT JOIN dbo.WN_HIK_Groups g WITH (NOLOCK) ON g.Id = d.Group_id
    WHERE @DeviceId IS NULL OR d.Id = @DeviceId
    ORDER BY CASE WHEN LOWER(g.Name) LIKE ''entrance%'' THEN 0 WHEN d.Device_Name LIKE ''Meeting%'' THEN 1 ELSE 2 END,
             TRY_CAST(d.Code AS INT), d.Device_Name, d.Id;
END');

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_DevCache_Save
    @DeviceId INT, @IsCards BIT, @Json NVARCHAR(MAX)
AS
BEGIN
    SET NOCOUNT ON;
    IF @IsCards = 1
        MERGE dbo.WN_HIK_DevCache WITH (HOLDLOCK) AS t USING (SELECT @DeviceId AS id) s ON t.Device_id = s.id
        WHEN MATCHED THEN UPDATE SET Cards_snapshot = @Json, Cards_UpdatedOn = SYSDATETIME()
        WHEN NOT MATCHED THEN INSERT (Device_id, Cards_snapshot, Cards_UpdatedOn) VALUES (s.id, @Json, SYSDATETIME());
    ELSE
        MERGE dbo.WN_HIK_DevCache WITH (HOLDLOCK) AS t USING (SELECT @DeviceId AS id) s ON t.Device_id = s.id
        WHEN MATCHED THEN UPDATE SET Users_snapshot = @Json, Users_UpdatedOn = SYSDATETIME()
        WHEN NOT MATCHED THEN INSERT (Device_id, Users_snapshot, Users_UpdatedOn) VALUES (s.id, @Json, SYSDATETIME());
END');

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_DevCache_Invalidate
    @DeviceId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_Events_GetLast
    @DeviceId INT

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_Event_Insert
    @DeviceId INT, @DeviceName NVARCHAR(100), @EmployeeNo NVARCHAR(32) = NULL, @Name NVARCHAR(128) = NULL,
    @CardNo NVARCHAR(32) = NULL, @AccessEvent INT = NULL, @AccessEventDetails NVARCHAR(64) = NULL,
    @MachineEventNo BIGINT = NULL, @EventTime DATETIME
AS
BEGIN
    SET NOCOUNT ON;
    -- Inserted = 0 when this machine event (device + serial + time) is already archived.
    IF @MachineEventNo IS NOT NULL AND EXISTS (SELECT 1 FROM dbo.WN_HIK_Events WITH (NOLOCK)
        WHERE device_id = @DeviceId AND machine_event_no = @MachineEventNo AND event_time = @EventTime)
    BEGIN
        SELECT 0 AS Inserted;
        RETURN;
    END
    BEGIN TRY
        INSERT INTO dbo.WN_HIK_Events (device_id, device_name, employee_no, name, card_no, access_event, access_event_details, machine_event_no, event_time)
        VALUES (@DeviceId, @DeviceName, @EmployeeNo, @Name, @CardNo, @AccessEvent, @AccessEventDetails, @MachineEventNo, @EventTime);
        SELECT 1 AS Inserted;
    END TRY
    BEGIN CATCH
        IF ERROR_NUMBER() IN (2601, 2627) SELECT 0 AS Inserted; ELSE THROW;
    END CATCH
END');

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_FpVault_Save
    @EmployeeNo NVARCHAR(32), @Name NVARCHAR(128), @FingerNo INT, @Template NVARCHAR(MAX)
AS
BEGIN
    SET NOCOUNT ON;
    MERGE dbo.WN_HIK_FpVault WITH (HOLDLOCK) AS t
    USING (SELECT @EmployeeNo AS emp, @Name AS nm, @FingerNo AS fno) s
       ON t.employee_no = s.emp AND t.name = s.nm AND t.finger_no = s.fno
    WHEN MATCHED THEN UPDATE SET template = @Template, updated_at = SYSDATETIME()
    WHEN NOT MATCHED THEN INSERT (employee_no, name, finger_no, template) VALUES (s.emp, s.nm, s.fno, @Template);
END');

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_Users_Rebuild
    @UsersJson NVARCHAR(MAX)
AS
BEGIN
    SET NOCOUNT ON; SET XACT_ABORT ON;
    -- Replaces WN_HIK_Users in one transaction from a JSON array built from the machine rosters.
    BEGIN TRAN;
        DELETE FROM dbo.WN_HIK_Users;
        INSERT INTO dbo.WN_HIK_Users (employee_no, name, room, role, machines, machine_count, cnic, tag_id)
        SELECT j.employee_no, j.name, j.room, j.role, j.machines, j.machine_count, j.cnic, j.tag_id
        FROM OPENJSON(@UsersJson) WITH (
            employee_no NVARCHAR(32) ''$.employee_no'', name NVARCHAR(128) ''$.name'', room NVARCHAR(256) ''$.room'',
            role NVARCHAR(16) ''$.role'', machines NVARCHAR(MAX) ''$.machines'', machine_count INT ''$.machine_count'',
            cnic NVARCHAR(20) ''$.cnic'', tag_id INT ''$.tag_id'') j;
    COMMIT;
END');

EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_HIK_Sync_Booking_GetNext
    @OldId INT, @CustomerCode NVARCHAR(50) = NULL, @SpaceId INT, @OldEnd DATETIME2(0)
AS
BEGIN
    SET NOCOUNT ON;
    -- Renewal: the earliest newer running booking of the same customer for the same space.
    SELECT TOP 1 b.Id, b.StartOn, b.EndOn, b.BillingPeriodEnd AS billEnd
    FROM dbo.WN_Bookings b WITH (NOLOCK)
    WHERE ISNULL(b.IsDeleted, 0) = 0 AND b.Id <> @OldId AND b.CustomerCode = @CustomerCode AND b.SpaceId = @SpaceId
      AND b.EndOn > @OldEnd AND b.EndOn > SYSDATETIME()
    ORDER BY b.StartOn ASC;
END');
