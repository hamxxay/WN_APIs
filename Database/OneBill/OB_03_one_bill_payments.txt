-- ============================================================================================
-- One Bill payment notifications: when a customer pays a One Bill invoice, EzPay calls the WorkNest API
-- (POST api/onebill/payment-notification) with { billno, rcvdamount, rcvddate, rcvdtime, rcvdvia }. Every notification
-- is kept in WN_OneBill_Payments, and the payment is recorded on the invoice with dbo.WN_RecordInvoicePayment (PaidTotal,
-- Paid / Partial status, access cards, WN_Payments row), all in one transaction.
-- Needs OB_01_one_bill (WN_Invoices.OneBill* columns) and INV_FIX_A (WN_RecordInvoicePayment).
-- Run as ONE statement in DBeaver: select all (Cmd+A), then Ctrl+Enter. No GO / USE needed.
-- One transaction: if anything fails, nothing is changed. Safe to run again.
-- ============================================================================================
SET XACT_ABORT ON;
BEGIN TRY
    BEGIN TRANSACTION;

    -- 1. Every notification EzPay sends, as received, with what was done with it.
    --    Outcome: Applied (payment recorded on the invoice), Duplicate (same notification again: not recorded twice),
    --    AlreadyPaid (invoice was already fully paid: kept for review, not recorded), Unmatched (no invoice has this bill
    --    number: kept for review), Invalid (amount missing or not above zero).
    IF OBJECT_ID('dbo.WN_OneBill_Payments', 'U') IS NULL
        EXEC(N'CREATE TABLE dbo.WN_OneBill_Payments (
            Id                 INT IDENTITY(1, 1) NOT NULL CONSTRAINT PK_WN_OneBill_Payments PRIMARY KEY,
            BillNo             VARCHAR(30)    NOT NULL,
            RcvdAmount         DECIMAL(18, 2) NOT NULL,
            RcvdDate           DATE           NULL,
            RcvdTime           TIME(0)        NULL,
            RcvdVia            NVARCHAR(50)   NULL,
            RawBody            NVARCHAR(MAX)  NULL,
            RemoteIp           VARCHAR(64)    NULL,
            ReceivedOn         DATETIME2(3)   NOT NULL CONSTRAINT DF_WN_OneBill_Payments_ReceivedOn DEFAULT (SYSUTCDATETIME()),
            InvoiceId          INT            NULL,
            Outcome            VARCHAR(20)    NOT NULL,
            Message            NVARCHAR(500)  NULL,
            PaidTotalAfter     DECIMAL(18, 4) NULL,
            InvoiceStatusAfter INT            NULL
        );');
    IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = 'IX_WN_OneBill_Payments_BillNo' AND object_id = OBJECT_ID('dbo.WN_OneBill_Payments'))
        EXEC(N'CREATE INDEX IX_WN_OneBill_Payments_BillNo ON dbo.WN_OneBill_Payments (BillNo, RcvdDate, RcvdTime, RcvdAmount);');
    IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = 'IX_WN_OneBill_Payments_InvoiceId' AND object_id = OBJECT_ID('dbo.WN_OneBill_Payments'))
        EXEC(N'CREATE INDEX IX_WN_OneBill_Payments_InvoiceId ON dbo.WN_OneBill_Payments (InvoiceId);');

    -- 2. Records one notification. One transaction: the notification row and the invoice payment are saved together
    --    or not at all (the API then answers an error and EzPay sends the notification again).
    EXEC(N'CREATE OR ALTER PROCEDURE dbo.WN_OneBill_RecordPayment
    @BillNo     VARCHAR(30),
    @RcvdAmount DECIMAL(18, 2),
    @RcvdDate   DATE          = NULL,
    @RcvdTime   TIME(0)       = NULL,
    @RcvdVia    NVARCHAR(50)  = NULL,
    @RawBody    NVARCHAR(MAX) = NULL,
    @RemoteIp   VARCHAR(64)   = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @Outcome VARCHAR(20), @Message NVARCHAR(500), @InvoiceId INT, @InvoiceNumber NVARCHAR(50),
            @Due DECIMAL(18, 4), @Paid DECIMAL(18, 4), @Matches INT, @PaidTotalAfter DECIMAL(18, 4), @StatusAfter INT,
            @NotificationId INT;
    SET @BillNo = LTRIM(RTRIM(@BillNo));

    BEGIN TRANSACTION;

    -- Same notification again (EzPay retries until it gets HTTP 200): answer it, record nothing twice.
    SELECT TOP 1 @NotificationId = p.Id, @InvoiceId = p.InvoiceId
      FROM dbo.WN_OneBill_Payments p WITH (UPDLOCK, HOLDLOCK)
     WHERE p.BillNo = @BillNo AND p.RcvdAmount = @RcvdAmount
       AND ISNULL(p.RcvdDate, ''19000101'') = ISNULL(@RcvdDate, ''19000101'')
       AND ISNULL(p.RcvdTime, ''00:00:00'') = ISNULL(@RcvdTime, ''00:00:00'')
       AND p.Outcome IN (''Applied'', ''AlreadyPaid'', ''Unmatched'')
     ORDER BY p.Id;

    IF @NotificationId IS NOT NULL
    BEGIN
        SET @Outcome = ''Duplicate'';
        SET @Message = ''Same notification as #'' + CAST(@NotificationId AS VARCHAR(12)) + ''; not recorded again.'';
    END
    ELSE IF ISNULL(@RcvdAmount, 0) <= 0
    BEGIN
        SET @Outcome = ''Invalid'';
        SET @Message = ''rcvdamount must be greater than 0.'';
    END
    ELSE
    BEGIN
        -- The invoice: EzPay sends our OneBillRef (e.g. 103400000459) as billno. The full One Bill number, or its last 12+
        -- digits (biller code + invoice Id), are matched as well in case a bank sends that instead.
        SELECT @Matches = COUNT(*), @InvoiceId = MIN(i.Id)
          FROM dbo.WN_Invoices i
         WHERE i.OneBillNumber IS NOT NULL
           AND (i.OneBillNumber = @BillNo OR i.OneBillRef = @BillNo
                OR (LEN(@BillNo) >= 12 AND RIGHT(i.OneBillNumber, LEN(@BillNo)) = @BillNo));

        IF ISNULL(@Matches, 0) <> 1
        BEGIN
            SET @InvoiceId = NULL;
            SET @Outcome = ''Unmatched'';
            SET @Message = CASE WHEN ISNULL(@Matches, 0) = 0 THEN ''No invoice has this bill number.''
                                ELSE ''More than one invoice matches this bill number.'' END;
        END
        ELSE
        BEGIN
            SELECT @InvoiceNumber = i.InvoiceNumber,
                   @Due = i.GrandTotal - ISNULL(i.WHTAmount, 0),
                   @Paid = ISNULL(i.PaidTotal, 0)
              FROM dbo.WN_Invoices i WHERE i.Id = @InvoiceId;

            IF @Paid >= @Due
            BEGIN
                SET @Outcome = ''AlreadyPaid'';
                SET @Message = ''Invoice '' + @InvoiceNumber + '' was already fully paid; payment kept for review, not recorded.'';
            END
            ELSE
            BEGIN
                -- Same procedure as Record Payment in the CMS: PaidTotal, Paid / Partial, access cards, WN_Payments row.
                DECLARE @Result TABLE (IsSuccessful INT, Message NVARCHAR(200), PaidTotal DECIMAL(18, 4), StatusId INT);
                DECLARE @Ref NVARCHAR(100) = N''1BILL-'' + @BillNo
                    + ISNULL(N''-'' + CONVERT(NVARCHAR(8), @RcvdDate, 112), N'''')
                    + ISNULL(N''-'' + REPLACE(CONVERT(NVARCHAR(8), @RcvdTime, 108), N'':'', N''''), N'''');
                DECLARE @Notes NVARCHAR(MAX) = N''Paid with 1Bill (EzPay)''
                    + ISNULL(N'' via '' + NULLIF(LTRIM(RTRIM(@RcvdVia)), N''''), N'''')
                    + ISNULL(N'' on '' + CONVERT(NVARCHAR(10), @RcvdDate, 23), N'''')
                    + ISNULL(N'' '' + CONVERT(NVARCHAR(8), @RcvdTime, 108), N'''')
                    + N'', bill no '' + @BillNo + N''.'';

                INSERT INTO @Result (IsSuccessful, Message, PaidTotal, StatusId)
                EXEC dbo.WN_RecordInvoicePayment
                     @InvoiceId = @InvoiceId, @PaidAmount = @RcvdAmount, @PaymentMethod = N''1Bill'',
                     @TransactionRef = @Ref, @Notes = @Notes;

                SELECT TOP 1 @PaidTotalAfter = PaidTotal, @StatusAfter = StatusId FROM @Result;
                SET @Outcome = ''Applied'';
                SET @Message = ''Payment of '' + CONVERT(VARCHAR(30), @RcvdAmount) + '' recorded on invoice '' + @InvoiceNumber
                    + CASE WHEN @PaidTotalAfter < @Due THEN '' (part payment).''
                           WHEN @PaidTotalAfter > @Due THEN '' (more than the amount due).''
                           ELSE ''.'' END;

                -- A payment proves the invoice is on EzPay, even if our call had been recorded as failed.
                UPDATE dbo.WN_Invoices
                   SET OneBillStatus = 2, OneBillCreatedOn = ISNULL(OneBillCreatedOn, GETDATE()), OneBillError = NULL
                 WHERE Id = @InvoiceId AND ISNULL(OneBillStatus, 0) <> 2;
            END;
        END;
    END;

    INSERT INTO dbo.WN_OneBill_Payments
        (BillNo, RcvdAmount, RcvdDate, RcvdTime, RcvdVia, RawBody, RemoteIp, InvoiceId, Outcome, Message, PaidTotalAfter, InvoiceStatusAfter)
    VALUES
        (@BillNo, ISNULL(@RcvdAmount, 0), @RcvdDate, @RcvdTime, @RcvdVia, @RawBody, @RemoteIp, @InvoiceId, @Outcome, LEFT(@Message, 500),
         @PaidTotalAfter, @StatusAfter);
    SET @NotificationId = SCOPE_IDENTITY();

    COMMIT TRANSACTION;

    SELECT @NotificationId AS NotificationId, @Outcome AS Outcome, @Message AS Message, @InvoiceId AS InvoiceId,
           @InvoiceNumber AS InvoiceNumber, @PaidTotalAfter AS PaidTotal, @StatusAfter AS StatusId;
END');

    COMMIT TRANSACTION;
    SELECT 'One Bill payments installed: WN_OneBill_Payments table and WN_OneBill_RecordPayment.' AS Result;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;
END CATCH;
