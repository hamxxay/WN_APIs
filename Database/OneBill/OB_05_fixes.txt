-- ============================================================================================
-- OB_05: One Bill fixes in dbo.WN_OneBill_ClaimInvoice, dbo.WN_OneBill_GetPending and dbo.WN_OneBill_RecordPayment.
--   1. Invoices awaiting approval (WN_Invoices.ApprovalStatus = 'PendingApproval') are never sent to EzPay.
--   2. WHT invoices (InvoiceTypeId 6 or WHTAmount > 0): EzPay gets the NET amount and every paid check uses it:
--        net due = ROUND(GrandTotal, 2) - WHT, WHT = ROUND(WHTAmount, 2), or ROUND(GrandTotal * WHTRate / 100, 2) when
--        WHTAmount is 0 (as dbo.WN_FeeReceipts_Insert); amount sent = net due - PaidTotal.
--      A payment that brings PaidTotal to the net due marks the invoice Paid and activates the booking's access cards
--      (WN_RecordInvoicePayment only does that once PaidTotal reaches GrandTotal, which a WHT invoice never does).
--   3. Deleted (StatusId 40), cancelled (OrderStatus 'Cancelled', legacy 5) and paid (OrderStatus 'Paid', legacy 2, or the
--      net due fully paid) invoices are not sent. A payment on a deleted / cancelled invoice answers the new
--      @StatusCode 106 (Outcome InvoiceClosed): kept in WN_OneBill_Payments for review, NOT recorded on the invoice.
--   4. Repeat notifications: a notification is the same as a recorded one only when bill number, amount, date AND time
--      are equal; when EzPay sent no date or no time, the raw body must be equal as well (two payments of the same
--      amount on the same bill without a time are no longer taken as one).
-- @StatusCode: 200 recorded; 201 same notification already recorded; 202 invoice already fully paid; 106 invoice
--              deleted / cancelled; 104 no invoice; 105 bad amount; 103 database error. The API answers success for 200 / 201.
-- Needs OB_01_one_bill, OB_03_one_bill_payments and OB_04_record_payment_status_code.
-- Run as ONE statement in DBeaver (select all, then Ctrl+Enter). One transaction. Safe to run again.
-- ============================================================================================
SET XACT_ABORT ON;
SET QUOTED_IDENTIFIER ON;
SET ANSI_NULLS ON;
BEGIN TRY
    IF OBJECT_ID('dbo.WN_OneBill_ClaimInvoice', 'P') IS NULL OR OBJECT_ID('dbo.WN_OneBill_GetPending', 'P') IS NULL
       OR COL_LENGTH('dbo.WN_Invoices', 'OneBillStatus') IS NULL
        THROW 50001, N'One Bill is not installed: run OB_01_one_bill.txt first. Nothing changed.', 1;
    IF OBJECT_ID('dbo.WN_OneBill_Payments', 'U') IS NULL
        THROW 50001, N'dbo.WN_OneBill_Payments is missing: run OB_03_one_bill_payments.txt first. Nothing changed.', 1;

    -- WN_Invoices.ApprovalStatus is missing on older databases: there no invoice awaits approval.
    DECLARE @Approval NVARCHAR(100) = CASE WHEN COL_LENGTH('dbo.WN_Invoices', 'ApprovalStatus') IS NULL THEN N'CAST(NULL AS NVARCHAR(50))' ELSE N'i.ApprovalStatus' END;
    DECLARE @Sql NVARCHAR(MAX);

    BEGIN TRANSACTION;

    -- 1. Claims an invoice for sending to EzPay and answers its data (Amount: what is left of the net due).
    --    Outcome: Claimed, Done (already created; OneBillNumber), Busy (being sent elsewhere), NotApplicable (no One Bill
    --    location, too old, awaiting approval, deleted, cancelled, paid, nothing due, or attempts used up).
    SET @Sql = N'CREATE OR ALTER PROCEDURE dbo.WN_OneBill_ClaimInvoice
    @InvoiceId    INT,
    @MaxAttempts  INT = 5,
    @StaleMinutes INT = 10
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @Outcome VARCHAR(20), @Status TINYINT, @Attempts INT, @ClaimedOn DATETIME, @Number VARCHAR(30),
            @BillerCode VARCHAR(10), @Prefix VARCHAR(20), @From DATETIME, @CreatedOn DATETIME,
            @StatusId INT, @Grand DECIMAL(18, 2), @Paid DECIMAL(18, 2), @LocationId INT,
            @Approval NVARCHAR(50), @Wht DECIMAL(18, 2), @NetDue DECIMAL(18, 2),
            @CancelledId INT = ISNULL((SELECT TOP 1 Id FROM dbo.OrderStatus WITH (NOLOCK) WHERE LTRIM(RTRIM(Description)) = ''Cancelled'' ORDER BY Id), 5),
            @PaidId INT = ISNULL((SELECT TOP 1 Id FROM dbo.OrderStatus WITH (NOLOCK) WHERE LTRIM(RTRIM(Description)) = ''Paid'' ORDER BY Id), 2);

    BEGIN TRANSACTION;

    -- WHT invoice (InvoiceTypeId 6 or WHTAmount > 0): WHTAmount, or GrandTotal * WHTRate / 100 (as WN_FeeReceipts_Insert).
    SELECT @Status = i.OneBillStatus, @Attempts = i.OneBillAttempts, @ClaimedOn = i.OneBillClaimedOn,
           @Number = i.OneBillNumber, @CreatedOn = i.CreatedOn, @StatusId = i.StatusId,
           @Grand = ROUND(ISNULL(i.GrandTotal, 0), 2), @Paid = ISNULL(i.PaidTotal, 0),
           @Wht = CASE WHEN ISNULL(i.WHTAmount, 0) > 0 THEN ROUND(i.WHTAmount, 2)
                       WHEN i.InvoiceTypeId = 6 THEN ROUND(ROUND(ISNULL(i.GrandTotal, 0), 2) * (ISNULL(i.WHTRate, 0) / 100.0), 2)
                       ELSE 0 END,
           @Approval = {ApprovalStatus},
           @LocationId = COALESCE(ci.LocationId, s.LocationId, u.LocationId)
      FROM dbo.WN_Invoices i WITH (UPDLOCK, HOLDLOCK)
      LEFT JOIN dbo.WN_CustomInvoices ci WITH (NOLOCK) ON ci.InvoiceId = i.Id
      LEFT JOIN dbo.WN_Bookings b WITH (NOLOCK) ON b.Id = i.BookingId
      LEFT JOIN dbo.WN_Spaces s WITH (NOLOCK) ON s.Id = b.SpaceId
      LEFT JOIN dbo.WN_Users u WITH (NOLOCK) ON u.Id = i.UserId
     WHERE i.Id = @InvoiceId;
    SET @NetDue = @Grand - ISNULL(@Wht, 0);

    SELECT @BillerCode = NULLIF(LTRIM(RTRIM(l.OneBillBillerCode)), ''''), @Prefix = NULLIF(LTRIM(RTRIM(l.OneBillPrefix)), ''''),
           @From = l.OneBillFrom
      FROM dbo.WN_Locations l WITH (NOLOCK)
     WHERE l.Id = @LocationId;

    IF @@ROWCOUNT = 0 OR @BillerCode IS NULL OR @Prefix IS NULL OR @CreatedOn IS NULL
       OR (@From IS NOT NULL AND @CreatedOn < @From) OR @InvoiceId > 99999999
        SET @Outcome = ''NotApplicable'';
    ELSE IF @Status = 2
        SET @Outcome = ''Done'';
    ELSE IF ISNULL(@StatusId, 0) IN (40, 5, 2, @CancelledId, @PaidId) OR ISNULL(@Approval, '''') = ''PendingApproval''
         OR @NetDue <= 0 OR @Paid >= @NetDue OR @Attempts >= @MaxAttempts
        SET @Outcome = ''NotApplicable'';
    ELSE IF @Status = 1 AND @ClaimedOn > DATEADD(MINUTE, -@StaleMinutes, GETDATE())
        SET @Outcome = ''Busy'';
    ELSE
    BEGIN
        SET @Outcome = ''Claimed'';
        UPDATE dbo.WN_Invoices
           SET OneBillStatus = 1,
               OneBillClaimedOn = GETDATE(),
               OneBillAttempts = OneBillAttempts + 1,
               OneBillRef = @BillerCode + RIGHT(REPLICATE(''0'', 8) + CAST(@InvoiceId AS VARCHAR(8)), 8),
               OneBillNumber = @Prefix + RIGHT(REPLICATE(''0'', 8) + CAST(@InvoiceId AS VARCHAR(8)), 8)
         WHERE Id = @InvoiceId;
    END;

    COMMIT TRANSACTION;

    IF @Outcome <> ''Claimed''
    BEGIN
        SELECT @Outcome AS Outcome, @InvoiceId AS InvoiceId, CASE WHEN @Outcome = ''Done'' THEN @Number END AS OneBillNumber;
        RETURN;
    END;

    SELECT TOP 1
        @Outcome AS Outcome,
        i.Id AS InvoiceId,
        i.InvoiceNumber,
        i.OneBillRef,
        i.OneBillNumber,
        CAST(@NetDue - @Paid AS DECIMAL(18, 2)) AS Amount,
        i.DueOn,
        i.BillingPeriodStart,
        i.BillingPeriodEnd,
        COALESCE(NULLIF(LTRIM(RTRIM(c.Code)), ''''), NULLIF(LTRIM(RTRIM(b.CustomerCode)), ''''), ''C'' + CAST(i.UserId AS VARCHAR(12))) AS RegistrationNumber,
        COALESCE(NULLIF(LTRIM(RTRIM(c.Company)), ''''), NULLIF(LTRIM(RTRIM(ISNULL(c.FirstName, '''') + '' '' + ISNULL(c.LastName, ''''))), ''''),
                 NULLIF(LTRIM(RTRIM(u.Name)), ''''), ''WorkNest Customer'') AS CustomerName,
        COALESCE(NULLIF(LTRIM(RTRIM(u.Email)), ''''), NULLIF(LTRIM(RTRIM(c.Email)), '''')) AS Email,
        COALESCE(NULLIF(LTRIM(RTRIM(c.PhoneNumber)), ''''), NULLIF(LTRIM(RTRIM(u.PhoneNumber)), '''')) AS Phone,
        l.Name AS LocationName
      FROM dbo.WN_Invoices i WITH (NOLOCK)
      LEFT JOIN dbo.WN_Bookings b WITH (NOLOCK) ON b.Id = i.BookingId
      LEFT JOIN dbo.WN_Users u WITH (NOLOCK) ON u.Id = i.UserId
      OUTER APPLY (SELECT TOP 1 cc.Code, cc.Company, cc.FirstName, cc.LastName, cc.Email, cc.PhoneNumber
                     FROM dbo.WN_Customers cc WITH (NOLOCK)
                    WHERE ((b.CustomerCode IS NOT NULL AND b.CustomerCode <> '''' AND cc.Code = b.CustomerCode)
                           OR (i.UserId > 0 AND cc.UserId = i.UserId))
                      AND (cc.IsActive = 1 OR cc.IsActive IS NULL)
                    ORDER BY CASE WHEN cc.Code = b.CustomerCode THEN 0 ELSE 1 END) c
      LEFT JOIN dbo.WN_Locations l WITH (NOLOCK) ON l.Id = @LocationId
     WHERE i.Id = @InvoiceId;
END';
    SET @Sql = REPLACE(@Sql, N'{ApprovalStatus}', @Approval);   -- see @Approval above
    EXEC(@Sql);

    -- 2. Invoices still to send: at a One Bill location, created on/after its start, not awaiting approval, not deleted /
    --    cancelled / paid, net due not fully paid, not yet created, attempts left, and not being sent right now.
    SET @Sql = N'CREATE OR ALTER PROCEDURE dbo.WN_OneBill_GetPending
    @Top          INT = 50,
    @MaxAttempts  INT = 5,
    @StaleMinutes INT = 10
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @CancelledId INT = ISNULL((SELECT TOP 1 Id FROM dbo.OrderStatus WITH (NOLOCK) WHERE LTRIM(RTRIM(Description)) = ''Cancelled'' ORDER BY Id), 5),
            @PaidId INT = ISNULL((SELECT TOP 1 Id FROM dbo.OrderStatus WITH (NOLOCK) WHERE LTRIM(RTRIM(Description)) = ''Paid'' ORDER BY Id), 2);
    SELECT TOP (@Top) i.Id AS InvoiceId
      FROM dbo.WN_Invoices i WITH (NOLOCK)
      LEFT JOIN dbo.WN_CustomInvoices ci WITH (NOLOCK) ON ci.InvoiceId = i.Id
      LEFT JOIN dbo.WN_Bookings b WITH (NOLOCK) ON b.Id = i.BookingId
      LEFT JOIN dbo.WN_Spaces s WITH (NOLOCK) ON s.Id = b.SpaceId
      LEFT JOIN dbo.WN_Users u WITH (NOLOCK) ON u.Id = i.UserId
      JOIN dbo.WN_Locations l WITH (NOLOCK) ON l.Id = COALESCE(ci.LocationId, s.LocationId, u.LocationId)
      -- Net due (as WN_OneBill_ClaimInvoice): GrandTotal less the WHT of a WHT invoice.
      CROSS APPLY (SELECT ROUND(ISNULL(i.GrandTotal, 0), 2)
                          - CASE WHEN ISNULL(i.WHTAmount, 0) > 0 THEN ROUND(i.WHTAmount, 2)
                                 WHEN i.InvoiceTypeId = 6 THEN ROUND(ROUND(ISNULL(i.GrandTotal, 0), 2) * (ISNULL(i.WHTRate, 0) / 100.0), 2)
                                 ELSE 0 END AS NetDue) n
     WHERE NULLIF(LTRIM(RTRIM(l.OneBillBillerCode)), '''') IS NOT NULL
       AND NULLIF(LTRIM(RTRIM(l.OneBillPrefix)), '''') IS NOT NULL
       AND (l.OneBillFrom IS NULL OR i.CreatedOn >= l.OneBillFrom)
       AND ISNULL(i.StatusId, 0) NOT IN (40, 5, 2, @CancelledId, @PaidId)
       AND ISNULL({ApprovalStatus}, '''') <> ''PendingApproval''
       AND n.NetDue > 0
       AND ISNULL(i.PaidTotal, 0) < n.NetDue
       AND i.OneBillAttempts < @MaxAttempts
       AND (i.OneBillStatus IS NULL OR i.OneBillStatus = 3
            OR (i.OneBillStatus = 1 AND i.OneBillClaimedOn <= DATEADD(MINUTE, -@StaleMinutes, GETDATE())))
     ORDER BY i.Id;
END';
    SET @Sql = REPLACE(@Sql, N'{ApprovalStatus}', @Approval);   -- see @Approval above
    EXEC(@Sql);

    -- 3. Records one EzPay notification (OB_04, with the fixes above).
    SET @Sql = N'CREATE OR ALTER PROCEDURE dbo.WN_OneBill_RecordPayment
    @ezpayRefNo      NVARCHAR(30),
    @ezpayRcvdAmount DECIMAL(18, 2),
    @ezpayRcvdDate   DATE          = NULL,
    @ezpayRcvdTime   TIME          = NULL,
    @ezpayRcvdVia    NVARCHAR(50)  = NULL,
    @StatusCode      INT OUTPUT,             -- 200 recorded, 201 already recorded, 202 invoice already paid, 106 invoice deleted / cancelled, 104 no invoice, 105 bad amount, 103 db error
    @RawBody         NVARCHAR(MAX) = NULL,
    @RemoteIp        VARCHAR(64)   = NULL
AS
BEGIN
    SET XACT_ABORT, NOCOUNT ON;

    DECLARE @Outcome VARCHAR(20), @Message NVARCHAR(500), @InvoiceId INT, @InvoiceNumber NVARCHAR(50), @Matches INT,
            @Due DECIMAL(18, 4), @Paid DECIMAL(18, 4), @PaidTotalAfter DECIMAL(18, 4), @StatusAfter INT,
            @Ref NVARCHAR(100), @Notes NVARCHAR(MAX), @NotificationId INT, @returnCode INT,
            @DuplicateId INT, @InvoiceStatusId INT, @BookingId INT, @BillingPeriodEnd DATETIME2,
            @CancelledId INT = ISNULL((SELECT TOP 1 Id FROM dbo.OrderStatus WITH (NOLOCK) WHERE LTRIM(RTRIM(Description)) = ''Cancelled'' ORDER BY Id), 5),
            @PaidId INT = ISNULL((SELECT TOP 1 Id FROM dbo.OrderStatus WITH (NOLOCK) WHERE LTRIM(RTRIM(Description)) = ''Paid'' ORDER BY Id), 2);
    DECLARE @Result TABLE (IsSuccessful INT, [Message] NVARCHAR(MAX), PaidTotal DECIMAL(18, 4), StatusId INT);

    SET @StatusCode = 200;
    SET @ezpayRefNo = LTRIM(RTRIM(@ezpayRefNo));

    BEGIN TRY
        BEGIN TRANSACTION;

        -- One notification at a time (EzPay can send the same one again while the first is still running).
        EXEC @returnCode = sp_getapplock @Resource = ''WN_OneBill_RecordPayment'', @LockMode = ''Exclusive'', @LockOwner = ''Transaction'';
        IF @returnCode < 0
            THROW 50020, N''Could not lock WN_OneBill_RecordPayment.'', 1;

        -- Same notification as one already recorded: same bill, amount, date and time. Without a date or a time, two
        -- payments of the same amount cannot be told apart that way, so the raw body must be the same as well.
        SELECT TOP 1 @DuplicateId = p.Id, @InvoiceId = p.InvoiceId, @PaidTotalAfter = p.PaidTotalAfter, @StatusAfter = p.InvoiceStatusAfter
          FROM dbo.WN_OneBill_Payments p
         WHERE p.BillNo = @ezpayRefNo AND p.RcvdAmount = @ezpayRcvdAmount AND p.Outcome = ''Applied''
           AND ((@ezpayRcvdDate IS NOT NULL AND @ezpayRcvdTime IS NOT NULL
                 AND p.RcvdDate = @ezpayRcvdDate AND p.RcvdTime = CAST(@ezpayRcvdTime AS TIME(0)))
                OR ((@ezpayRcvdDate IS NULL OR @ezpayRcvdTime IS NULL)
                    AND ISNULL(p.RcvdDate, ''19000101'') = ISNULL(@ezpayRcvdDate, ''19000101'')
                    AND ISNULL(p.RcvdTime, ''00:00:00'') = ISNULL(CAST(@ezpayRcvdTime AS TIME(0)), ''00:00:00'')
                    AND ISNULL(p.RawBody, N'''') = ISNULL(@RawBody, N'''')))
         ORDER BY p.Id;

        IF ISNULL(@ezpayRcvdAmount, 0) <= 0
        BEGIN
            SELECT @InvoiceId = NULL, @PaidTotalAfter = NULL, @StatusAfter = NULL;
            SELECT @StatusCode = 105, @Outcome = ''Invalid'', @Message = ''Amount missing or not above zero.'';
        END
        ELSE IF @DuplicateId IS NOT NULL
        BEGIN
            -- Nothing to do (EzPay just did not get our first answer).
            SELECT @StatusCode = 201, @Outcome = ''Duplicate'', @Message = ''Same notification as one already recorded; not recorded again.'';
        END
        ELSE
        BEGIN
            -- The invoice: the 1-Link consumer number (OneBillNumber) or our ref (OneBillRef).
            SELECT @Matches = COUNT(*), @InvoiceId = MIN(i.Id)
              FROM dbo.WN_Invoices i
             WHERE i.OneBillNumber = @ezpayRefNo OR i.OneBillRef = @ezpayRefNo;

            IF ISNULL(@Matches, 0) <> 1
                SELECT @InvoiceId = NULL, @StatusCode = 104, @Outcome = ''Unmatched'',
                       @Message = CASE WHEN ISNULL(@Matches, 0) = 0 THEN ''No invoice has this bill number.'' ELSE ''More than one invoice has this bill number.'' END;
            ELSE
            BEGIN
                -- Net due: GrandTotal less the WHT of a WHT invoice (InvoiceTypeId 6 or WHTAmount > 0), as WN_FeeReceipts_Insert.
                SELECT @InvoiceNumber = i.InvoiceNumber,
                       @Due  = ROUND(ISNULL(i.GrandTotal, 0), 2)
                               - CASE WHEN ISNULL(i.WHTAmount, 0) > 0 THEN ROUND(i.WHTAmount, 2)
                                      WHEN i.InvoiceTypeId = 6 THEN ROUND(ROUND(ISNULL(i.GrandTotal, 0), 2) * (ISNULL(i.WHTRate, 0) / 100.0), 2)
                                      ELSE 0 END,
                       @Paid = ISNULL(i.PaidTotal, 0),
                       @InvoiceStatusId = i.StatusId,
                       @BookingId = i.BookingId,
                       @BillingPeriodEnd = ISNULL(i.BillingPeriodEnd, b.EndOn)
                  FROM dbo.WN_Invoices i WITH (UPDLOCK, HOLDLOCK)
                  LEFT JOIN dbo.WN_Bookings b WITH (NOLOCK) ON b.Id = i.BookingId
                 WHERE i.Id = @InvoiceId;

                IF ISNULL(@InvoiceStatusId, 0) IN (40, 5, @CancelledId)
                    SELECT @StatusCode = 106, @Outcome = ''InvoiceClosed'',
                           @Message = ''Invoice '' + @InvoiceNumber + '' is ''
                                      + CASE WHEN @InvoiceStatusId = 40 THEN ''deleted'' ELSE ''cancelled'' END
                                      + ''; payment not recorded (refund it or record it on another invoice).'';
                ELSE IF @Paid >= @Due OR ISNULL(@InvoiceStatusId, 0) IN (2, @PaidId)
                    SELECT @StatusCode = 202, @Outcome = ''AlreadyPaid'',
                           @Message = ''Invoice '' + @InvoiceNumber + '' was already fully paid; payment not recorded (check for a double payment).'';
                ELSE
                BEGIN
                    SET @Ref = N''1BILL-'' + @ezpayRefNo
                             + ISNULL(N''-'' + CONVERT(NVARCHAR(8), @ezpayRcvdDate, 112), N'''')
                             + ISNULL(N''-'' + REPLACE(CONVERT(NVARCHAR(8), @ezpayRcvdTime, 108), N'':'', N''''), N'''');
                    SET @Notes = N''Paid with 1Bill (EzPay)''
                             + ISNULL(N'' via '' + NULLIF(LTRIM(RTRIM(@ezpayRcvdVia)), N''''), N'''')
                             + ISNULL(N'' on '' + CONVERT(NVARCHAR(10), @ezpayRcvdDate, 23), N'''')
                             + ISNULL(N'' '' + CONVERT(NVARCHAR(8), @ezpayRcvdTime, 108), N'''')
                             + N'', bill no '' + @ezpayRefNo + N''.'';

                    INSERT INTO @Result (IsSuccessful, [Message], PaidTotal, StatusId)
                    EXEC dbo.WN_RecordInvoicePayment
                         @InvoiceId = @InvoiceId, @PaidAmount = @ezpayRcvdAmount, @PaymentMethod = N''1Bill'',
                         @TransactionRef = @Ref, @Notes = @Notes;

                    SELECT TOP 1 @PaidTotalAfter = PaidTotal, @StatusAfter = StatusId FROM @Result;
                    IF @PaidTotalAfter IS NULL OR @PaidTotalAfter < @Paid + @ezpayRcvdAmount
                        THROW 50010, N''WN_RecordInvoicePayment did not record the payment.'', 1;   -- all rolled back, 103

                    -- Net due paid: Paid, and the access cards active, as WN_RecordInvoicePayment does once GrandTotal is
                    -- paid (a WHT invoice never reaches GrandTotal: the customer pays the net, the WHT is paid to the FBR).
                    IF @PaidTotalAfter >= @Due AND ISNULL(@StatusAfter, 0) <> @PaidId
                    BEGIN
                        UPDATE dbo.WN_Invoices SET StatusId = @PaidId WHERE Id = @InvoiceId;
                        SET @StatusAfter = @PaidId;
                        IF @BookingId IS NOT NULL AND @BookingId > 0
                            UPDATE dbo.WN_AccessCards
                               SET Status = 1, -- 1 = Active
                                   EndDate = ISNULL(@BillingPeriodEnd, EndDate),
                                   UpdatedOn = SYSUTCDATETIME()
                             WHERE BookingId = @BookingId;
                    END;

                    SET @Outcome = ''Applied'';
                    SET @Message = ''Payment of '' + CONVERT(VARCHAR(30), @ezpayRcvdAmount) + '' recorded on invoice '' + @InvoiceNumber
                        + CASE WHEN @PaidTotalAfter < @Due THEN '' (part payment).''
                               WHEN @PaidTotalAfter > @Due THEN '' (more than the amount due).''
                               ELSE ''.'' END;

                    -- A payment proves the invoice is on EzPay, even if our call had been recorded as failed.
                    UPDATE dbo.WN_Invoices
                       SET OneBillStatus = 2, OneBillCreatedOn = ISNULL(OneBillCreatedOn, GETDATE()), OneBillError = NULL
                     WHERE Id = @InvoiceId AND ISNULL(OneBillStatus, 0) <> 2;
                END;
            END;
        END;

        -- Every notification is kept, whatever its outcome (rows that were not recorded are for review).
        INSERT INTO dbo.WN_OneBill_Payments
            (BillNo, RcvdAmount, RcvdDate, RcvdTime, RcvdVia, RawBody, RemoteIp, InvoiceId, Outcome, [Message], PaidTotalAfter, InvoiceStatusAfter)
        VALUES
            (LEFT(@ezpayRefNo, 30), ISNULL(@ezpayRcvdAmount, 0), @ezpayRcvdDate, CAST(@ezpayRcvdTime AS TIME(0)), @ezpayRcvdVia, @RawBody, @RemoteIp,
             @InvoiceId, @Outcome, LEFT(@Message, 500), @PaidTotalAfter, @StatusAfter);
        SET @NotificationId = SCOPE_IDENTITY();

        COMMIT TRANSACTION;

        SELECT @NotificationId AS NotificationId, @StatusCode AS StatusCode, @Outcome AS Outcome, @Message AS Message,
               @InvoiceId AS InvoiceId, @InvoiceNumber AS InvoiceNumber, @PaidTotalAfter AS PaidTotal, @StatusAfter AS StatusId;
    END TRY
    BEGIN CATCH
        SET @StatusCode = 103;   -- database error: nothing kept, EzPay should send it again
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END';
    EXEC(@Sql);

    COMMIT TRANSACTION;
    SELECT 'OB_05 installed: One Bill skips invoices awaiting approval / deleted / cancelled / paid, uses the net of WHT, answers 106 for closed invoices.' AS Result;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;
END CATCH;
